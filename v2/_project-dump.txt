        // Загружаем хеш пароля из Firebase
        async function getPasswordHash() {
            try {
                const docRef = window.firestore.doc(window.db, 'auth', 'password');
                const docSnap = await window.firestore.getDoc(docRef);
                if (docSnap.exists()) {
                    return docSnap.data().hash;
                }
                return null;
            } catch (error) {
                console.error('Ошибка загрузки хеша:', error);
                return null;
            }
        }

        // Проверка пароля
        async function checkPassword(enteredPassword) {
            const correctHash = await getPasswordHash();
            if (!correctHash) {
                alert('Ошибка загрузки данных авторизации');
                return false;
            }

            // Хешируем введённый пароль и сравниваем
            const encoder = new TextEncoder();
            const data = encoder.encode(enteredPassword);
            const hashBuffer = await crypto.subtle.digest('SHA-256', data);
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            const enteredHash = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

            return enteredHash === correctHash;
        }

        document.getElementById('loginButton').addEventListener('click', async () => {
            const entered = document.getElementById('passwordInput').value;
            const isValid = await checkPassword(entered);

            if (isValid) {
                sessionStorage.setItem('vacationAuth', 'true');
                window.location.href = 'index.html';
            } else {
                document.getElementById('errorText').style.display = 'block';
            }
        });

        // Вход по Enter
        document.getElementById('passwordInput').addEventListener('keypress', async (e) => {
            if (e.key === 'Enter') {
                document.getElementById('loginButton').click();
            }
        });

// --- Утилиты ---
const getCalendarDays = (startStr, endStr) => {
    if (!startStr || !endStr) return 0;
    const start = new Date(startStr);
    const end = new Date(endStr);
    if (end < start) return 0;
    return Math.floor((end - start) / (1000 * 3600 * 24)) + 1;
};

const getWorkingDays = (startStr, endStr) => {
    if (!startStr || !endStr) return 0;
    let count = 0;
    const start = new Date(startStr);
    const end = new Date(endStr);
    if (end < start) return 0;
    for (let d = new Date(start); d <= end; d.setDate(d.getDate() + 1)) {
        const dayOfWeek = d.getDay();
        if (dayOfWeek !== 0 && dayOfWeek !== 6) count++;
    }
    return count;
};

const validateInternalOverlap = (user, currentVacId) => {
    const vacs = user.vacations || [];
    const target = vacs.find(v => v.id === currentVacId);
    if (!target) return { isOverlap: false, isInvalid: false };

    if (target.startDate && target.endDate && target.endDate < target.startDate) {
        return { isOverlap: false, isInvalid: true };
    }

    const isOverlap = vacs.some(v => {
        if (v.id === currentVacId) return false;
        if (!v.startDate || !v.endDate || !target.startDate || !target.endDate) return false;

        const vStart = parseInt(v.startDate.replace(/-/g, ''), 10);
        const vEnd = parseInt(v.endDate.replace(/-/g, ''), 10);
        const tStart = parseInt(target.startDate.replace(/-/g, ''), 10);
        const tEnd = parseInt(target.endDate.replace(/-/g, ''), 10);

        return tStart <= vEnd && vStart <= tEnd;
    });

    return { isOverlap, isInvalid: false };
};

const getExternalOverlaps = (currentUser, currentVac) => {
    if (!currentVac.startDate || !currentVac.endDate) return [];
    if (currentVac.endDate < currentVac.startDate) return [];

    const tStart = parseInt(currentVac.startDate.replace(/-/g, ''), 10);
    const tEnd = parseInt(currentVac.endDate.replace(/-/g, ''), 10);
    const overlaps = [];

    state.users.forEach(otherUser => {
        if (otherUser.id === currentUser.id) return;
        if (otherUser.isVisible === false) return;

        (otherUser.vacations || []).forEach(otherVac => {
            if (!otherVac.startDate || !otherVac.endDate) return;

            const oStart = parseInt(otherVac.startDate.replace(/-/g, ''), 10);
            const oEnd = parseInt(otherVac.endDate.replace(/-/g, ''), 10);

            if (tStart <= oEnd && oStart <= tEnd) {
                if (!overlaps.find(o => o.name === otherUser.name)) {
                    overlaps.push({ name: otherUser.name, color: otherUser.color });
                }
            }
        });
    });

    return overlaps;
};

const countDaysFromYearStart = (dateStr) => {
    if (!dateStr) return 0;
    const d = new Date(dateStr);
    const yearStart = new Date(state.year, 0, 1);
    return Math.floor((d - yearStart) / (1000 * 3600 * 24)) + 1;
};

            animation: 150,
            handle: '.form__drag-handle',

            }
        });
    }

function updateCalendarVacations() {

            if (vac.endDate < vac.startDate) return;

            const startIndex = countDaysFromYearStart(vac.startDate);

                document.querySelectorAll('.calendar__day--highlighted').forEach(d => d.classList.remove('calendar__day--highlighted'));

                    const d = document.getElementById(`dayid_${i}`);

import { state } from './state.js';

let selectionStart = null;

export function initCalendarInteractions() {
    const calendarContainer = document.getElementById('calendar-container');
    if (!calendarContainer) return;

    calendarContainer.addEventListener('click', (e) => {
        const dayDiv = e.target.closest('.calendar__day[id^="dayid"]');
        if (!dayDiv) return;

        const dayIndex = parseInt(dayDiv.id.replace('dayid_', ''), 10);
        if (!dayIndex) return;

        if (selectionStart === null) {
            selectionStart = dayIndex;
            highlightRange(selectionStart, selectionStart);
            return;
        }

        let start = Math.min(selectionStart, dayIndex);
        let end = Math.max(selectionStart, dayIndex);

        highlightRange(start, end);

        const yearStart = new Date(state.year, 0, 1);

        const sDate = new Date(yearStart);
        sDate.setDate(sDate.getDate() + start - 1);
        const sStr = `${sDate.getFullYear()}-${String(sDate.getMonth() + 1).padStart(2, '0')}-${String(sDate.getDate()).padStart(2, '0')}`;

        const eDate = new Date(yearStart);
        eDate.setDate(eDate.getDate() + end - 1);
        const eStr = `${eDate.getFullYear()}-${String(eDate.getMonth() + 1).padStart(2, '0')}-${String(eDate.getDate()).padStart(2, '0')}`;

        openCreateVacationPopup(sStr, eStr, start, end);
        selectionStart = null;
    });
}

function highlightRange(startIndex, endIndex) {
    document.querySelectorAll('.calendar__day.selected').forEach(el => el.classList.remove('selected'));
    for (let i = startIndex; i <= endIndex; i++) {
        const dayDiv = document.getElementById(`dayid_${i}`);
        if (dayDiv) dayDiv.classList.add('selected');
    }
}

function clearSelection() {
    document.querySelectorAll('.calendar__day.selected').forEach(el => el.classList.remove('selected'));
    selectionStart = null;
}

// Функция для поиска отпусков в выбранном диапазоне
function getVacationsInRange(startIdx, endIdx) {
    const result = [];
    state.users.forEach(user => {
        (user.vacations || []).forEach(vac => {
            if (!vac.startDate || !vac.endDate) return;
            const dStart = new Date(vac.startDate);
            const dEnd = new Date(vac.endDate);
            const yearStart = new Date(state.year, 0, 1);
            const vStart = Math.floor((dStart - yearStart) / (1000 * 3600 * 24)) + 1;
            const vEnd = Math.floor((dEnd - yearStart) / (1000 * 3600 * 24)) + 1;

            if (startIdx <= vEnd && endIdx >= vStart) {
                result.push({ user, vac });
            }
        });
    });
    return result;
}

// scripts/calendar.js — v2

const months = {
    0: "Январь", 1: "Февраль", 2: "Март", 3: "Апрель",
    4: "Май", 5: "Июнь", 6: "Июль", 7: "Август",
    8: "Сентябрь", 9: "Октябрь", 10: "Ноябрь", 11: "Декабрь",
};

function createMonth(monthName) {
    const monthDiv = document.createElement("div");
    monthDiv.classList.add("calendar__month");
    const title = document.createElement("h2");
    title.classList.add("calendar__title");
    title.textContent = monthName;
    const area = document.createElement("div");
    area.classList.add("calendar__area");
    monthDiv.append(title, area);
    return monthDiv;
}

function createDay(dayNumber) {
    const dayDiv = document.createElement("div");
    dayDiv.classList.add("calendar__day");
    if (dayNumber !== "") dayDiv.textContent = dayNumber;
    return dayDiv;
}

export function renderCalendar(year) {
    const calendarContainer = document.querySelector(".calendar");
    if (!calendarContainer) return;

    calendarContainer.innerHTML = '';
    const startDate = new Date(year, 0, 1);
    const endDate = new Date(year, 11, 31);

    drawCalendar(calendarContainer, startDate, endDate);

    fetch(`https://isdayoff.ru/api/getdata?year=${year}`, { signal: AbortSignal.timeout(3000) })
        .then(response => response.text())
        .then(data => applyHolidays(data.split("")))
        .catch(err => console.warn("API isdayoff.ru недоступен.", err));
}

function drawCalendar(container, startDate, endDate) {
    let dayIndex = 0;
    let currentMonthDiv = null;

    for (let d = new Date(startDate); d <= endDate; d.setDate(d.getDate() + 1)) {
        const dayOfMonth = d.getDate();
        const monthNumber = d.getMonth();
        const dayOfWeek = d.getDay();

        if (dayOfMonth === 1) {
            currentMonthDiv = createMonth(months[monthNumber]);
            container.append(currentMonthDiv);

            // Сдвиг для первого дня (пустые ячейки)
            const startCol = dayOfWeek === 0 ? 7 : dayOfWeek;
            for (let i = 1; i < startCol; i++) {
                currentMonthDiv.querySelector(".calendar__area").append(createDay(""));
            }
        }

        const dayDiv = createDay(dayOfMonth);

        if (dayOfWeek === 0 || dayOfWeek === 6) {
            dayDiv.classList.add("calendar__day_dayoff");
        }

        dayDiv.id = `dayid_${dayIndex + 1}`;

        currentMonthDiv.querySelector(".calendar__area").append(dayDiv);
        dayIndex++;
    }

    window.dispatchEvent(new Event("calendarReady"));
}

function applyHolidays(prodCalList) {
    for (let i = 0; i < prodCalList.length; i++) {
        if (prodCalList[i] === "1") {
            const dayDiv = document.getElementById(`dayid_${i + 1}`);
            if (dayDiv && !dayDiv.classList.contains("calendar__day_dayoff")) {
                dayDiv.classList.add("calendar__day_dayoff");
            }
        }
    }
}

// state.js — v2 (Event-Driven + фикс onSnapshot)

const MIN_YEAR = 2000;
const MAX_YEAR = 2100;

function isValidYear(y) { return Number.isInteger(y) && y >= MIN_YEAR && y <= MAX_YEAR; }

function resolveInitialYear() {
    const params = new URLSearchParams(window.location.search);
    const urlYear = Number(params.get("year"));
    if (isValidYear(urlYear)) return urlYear;
    const lsYear = Number(localStorage.getItem("vacationYear"));
    if (isValidYear(lsYear)) return lsYear;
    return new Date().getFullYear();
}

export const state = {
    year: resolveInitialYear(),
    users: [],
    loading: true,
    error: null,
    _unsubscribe: null,
    _pendingRemote: null,
    _localEditing: false, // Флаг блокировки входящих обновлений
};

export const stateEvents = new EventTarget();

function emitChange() {
    stateEvents.dispatchEvent(new Event('statechange'));
}

function usersCollection(year) {
    return window.firestore.collection(window.db, "vacationData_v2", String(year), "users");
}

function isUserTypingNow() {
    const el = document.activeElement;
    if (!el) return false;
    if (["INPUT", "TEXTAREA", "SELECT"].includes(el.tagName)) return true;
    if (el.isContentEditable) return true;
    return false;
}

function applyRemoteUsers(users) {
    if (isUserTypingNow() || state._localEditing) {
        state._pendingRemote = users;
        return;
    }
    state.users = users;
    state.loading = false;
    emitChange();
}

function flushPendingRemote() {
    if (state._pendingRemote) {
        state.users = state._pendingRemote;
        state._pendingRemote = null;
        emitChange();
    }
}

document.addEventListener("focusout", () => setTimeout(flushPendingRemote, 50), true);

async function startSubscription(year) {
    if (state._unsubscribe) state._unsubscribe();
    await window.firebaseReady;

    state.year = year;
    state.loading = true;
    state.users = [];
    emitChange();

    const { onSnapshot, query } = window.firestore;
    state._unsubscribe = onSnapshot(query(usersCollection(year)), (snapshot) => {
        const users = [];
        snapshot.forEach(doc => {
            const data = doc.data();
            if (!data.order) data.order = 0;
            users.push({ id: doc.id, ...data });
        });

        users.sort((a, b) => a.order - b.order);

        // Если мы сейчас сохраняем локальные изменения — игнорируем
        if (state._localEditing) {
            console.log('[state] localEditing, skipping snapshot');
            return;
        }

        applyRemoteUsers(users);
    }, (error) => {
        console.error("[state] Ошибка синхронизации:", error);
        state.error = error.message;
        state.loading = false;
        emitChange();
    });
}

export async function initState() {
    const year = state.year;
    localStorage.setItem("vacationYear", String(year));
    const url = new URL(window.location.href);
    url.searchParams.set("year", String(year));
    window.history.replaceState({}, "", url);

    await startSubscription(year);
}

export async function setYear(newYear) {
    if (newYear === state.year) return;
    localStorage.setItem("vacationYear", String(newYear));
    const url = new URL(window.location.href);
    url.searchParams.set("year", String(newYear));
    window.history.pushState({}, "", url);
    await startSubscription(newYear);
}

export function getAvailableYears() {
    const currentYear = new Date().getFullYear();
    const years = [];
    for (let y = currentYear - 2; y <= currentYear + 3; y++) {
        years.push(y);
    }
    return years;
}
