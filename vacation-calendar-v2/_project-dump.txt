        // Загружаем хеш пароля из Firebase
        async function getPasswordHash() {
            try {
                const docRef = window.firestore.doc(window.db, 'auth', 'password');
                const docSnap = await window.firestore.getDoc(docRef);
                if (docSnap.exists()) {
                    return docSnap.data().hash;
                }
                return null;
            } catch (error) {
                console.error('Ошибка загрузки хеша:', error);
                return null;
            }
        }

        // Проверка пароля
        async function checkPassword(enteredPassword) {
            const correctHash = await getPasswordHash();
            if (!correctHash) {
                alert('Ошибка загрузки данных авторизации');
                return false;
            }

            // Хешируем введённый пароль и сравниваем
            const encoder = new TextEncoder();
            const data = encoder.encode(enteredPassword);
            const hashBuffer = await crypto.subtle.digest('SHA-256', data);
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            const enteredHash = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

            return enteredHash === correctHash;
        }

        document.getElementById('loginButton').addEventListener('click', async () => {
            const entered = document.getElementById('passwordInput').value;
            const isValid = await checkPassword(entered);

            if (isValid) {
                sessionStorage.setItem('vacationAuth', 'true');
                window.location.href = 'index.html';
            } else {
                document.getElementById('errorText').style.display = 'block';

            if (e.key === 'Enter') {

function updateCalendarVacations() {
    document.querySelectorAll('.calendar__span').forEach(el => el.remove());
    const yearStart = new Date(state.year, 0, 1);

    state.users.forEach(user => {
        (user.vacations || []).forEach(vac => {
            if (!vac.startDate || !vac.endDate) return;
            const start = new Date(vac.startDate);
            const end = new Date(vac.endDate);

            // Вычисляем индексы дней
            const startIndex = Math.floor((start - yearStart) / (1000 * 3600 * 24)) + 1;
            const endIndex = Math.floor((end - yearStart) / (1000 * 3600 * 24)) + 1;

            for (let i = startIndex; i <= endIndex; i++) {
                const dayDiv = document.getElementById(`dayid_${i}`);
                if (dayDiv) {
                    const span = document.createElement('span');
                    span.classList.add('calendar__span');
                    span.style.backgroundColor = user.color;
                    // Чтобы точки были видны, зададим им размер (стили могут перебивать)
                    span.style.width = '10px';
                    span.style.height = '10px';
                    span.style.borderRadius = '50%';
                    span.style.display = 'inline-block';
                    span.style.margin = '2px';
                    dayDiv.appendChild(span);
                }
            }
        });
    });
}

// scripts/calendar.js — v2

const months = {
    0: "Январь", 1: "Февраль", 2: "Март", 3: "Апрель",
    4: "Май", 5: "Июнь", 6: "Июль", 7: "Август",
    8: "Сентябрь", 9: "Октябрь", 10: "Ноябрь", 11: "Декабрь",
};

function createMonth(monthName) {
    const monthDiv = document.createElement("div");
    monthDiv.classList.add("calendar__month");

    const title = document.createElement("h2");
    title.classList.add("calendar__title");
    title.textContent = monthName;

    const area = document.createElement("div");
    area.classList.add("calendar__area");

    monthDiv.append(title, area);
    return monthDiv;
}

function createDay(dayNumber) {
    const dayDiv = document.createElement("div");
    dayDiv.classList.add("calendar__day");
    if (dayNumber !== "") {
        dayDiv.textContent = dayNumber;
    }
    return dayDiv;
}

export function renderCalendar(year) {
    const calendarContainer = document.querySelector(".calendar");
    if (!calendarContainer) return;

    calendarContainer.innerHTML = '';

    const startDate = new Date(year, 0, 1);
    const endDate = new Date(year, 11, 31);

    fetch(`https://isdayoff.ru/api/getdata?year=${year}`)
        .then(response => response.text())
        .then(data => {
            const prodCalList = data.split("");
            drawCalendar(calendarContainer, startDate, endDate, prodCalList);
        })
        .catch(err => {
            console.error("Ошибка получения календаря:", err);
            const fallbackCal = "0000000".repeat(53).split("");
            drawCalendar(calendarContainer, startDate, endDate, fallbackCal);
        });
}

function drawCalendar(container, startDate, endDate, prodCalList) {
    let dayIndex = 0;
    let currentMonthDiv = null;

    for (let d = new Date(startDate); d <= endDate; d.setDate(d.getDate() + 1)) {
        const dayOfMonth = d.getDate();
        const monthNumber = d.getMonth();
        const dayOfWeek = d.getDay();

        if (dayOfMonth === 1) {
            currentMonthDiv = createMonth(months[monthNumber]);
            container.append(currentMonthDiv);

            const emptyDays = dayOfWeek > 0 ? dayOfWeek - 1 : 6;
            for (let i = 0; i < emptyDays; i++) {
                currentMonthDiv.querySelector(".calendar__area").append(createDay(""));
            }
        }

        const dayDiv = createDay(dayOfMonth);

        if (prodCalList[dayIndex] === "1") {
            dayDiv.classList.add("calendar__day_dayoff");
        }

        // ВОТ ОНО! Возвращаем ID для каждого дня
        dayDiv.id = `dayid_${dayIndex + 1}`;

        currentMonthDiv.querySelector(".calendar__area").append(dayDiv);
        dayIndex++;
    }

    window.dispatchEvent(new Event("calendarReady"));
}

  if (hexColorIndexes[0] >= 10 && hexColorIndexes[2] >= 10 && hexColorIndexes[4] >= 10) {

function checkContentEditableLength(event) {

// state.js — v2 (Event-Driven + Локальная сортировка)
const MIN_YEAR = 2000;
const MAX_YEAR = 2100;

function isValidYear(y) { return Number.isInteger(y) && y >= MIN_YEAR && y <= MAX_YEAR; }

function resolveInitialYear() {
    const params = new URLSearchParams(window.location.search);
    const urlYear = Number(params.get("year"));
    if (isValidYear(urlYear)) return urlYear;
    const lsYear = Number(localStorage.getItem("vacationYear"));
    if (isValidYear(lsYear)) return lsYear;
    return new Date().getFullYear();
}

export const state = {
    year: resolveInitialYear(),
    users: [],
    loading: true,
    error: null,
    _unsubscribe: null,
    _pendingRemote: null,
};

export const stateEvents = new EventTarget();

function emitChange() {
    stateEvents.dispatchEvent(new Event('statechange'));
}

function usersCollection(year) {
    // Убрали orderBy, чтобы не падало из-за старых записей
    return window.firestore.collection(window.db, "vacationData_v2", String(year), "users");
}

function isUserTypingNow() {
    const el = document.activeElement;
    if (!el) return false;
    if (["INPUT", "TEXTAREA", "SELECT"].includes(el.tagName)) return true;
    if (el.isContentEditable) return true;
    return false;
}

function applyRemoteUsers(users) {
    if (isUserTypingNow() || state._localEditing) {
        state._pendingRemote = users;
        return;
    }
    state.users = users;
    state.loading = false;
    emitChange();
}

function flushPendingRemote() {
    if (state._pendingRemote) {
        state.users = state._pendingRemote;
        state._pendingRemote = null;
        emitChange();
    }
}

document.addEventListener("focusout", () => setTimeout(flushPendingRemote, 50), true);

async function startSubscription(year) {
    if (state._unsubscribe) state._unsubscribe();
    await window.firebaseReady;

    state.year = year;
    state.loading = true;
    state.users = [];
    emitChange();

    const { onSnapshot, query } = window.firestore;
    state._unsubscribe = onSnapshot(query(usersCollection(year)), (snapshot) => {
        const users = [];
        snapshot.forEach(doc => {
            const data = doc.data();
            // Если у старого юзера нет order, ставим 0 (он будет первым)
            if (!data.order) data.order = 0;
            users.push({ id: doc.id, ...data });
        });

        // Локальная сортировка по order
        users.sort((a, b) => a.order - b.order);

        applyRemoteUsers(users);
    }, (error) => {
        console.error("[state] Ошибка синхронизации:", error);
        state.error = error.message;
        state.loading = false;
        emitChange();
    });
}

export async function initState() {
    const year = state.year;
    localStorage.setItem("vacationYear", String(year));
    const url = new URL(window.location.href);
    url.searchParams.set("year", String(year));
    window.history.replaceState({}, "", url);

    await startSubscription(year);
}

export async function setYear(newYear) {
    if (newYear === state.year) return;
    localStorage.setItem("vacationYear", String(newYear));
    const url = new URL(window.location.href);
    url.searchParams.set("year", String(newYear));
    window.history.pushState({}, "", url);
    await startSubscription(newYear);
}

  const errorElement = formElement.querySelector(`.${inputElement.name}-input-error`);
  inputElement.classList.add(settings.inputErrorClass);
  errorElement.textContent = inputElement.validationMessage;
  errorElement.classList.add(settings.errorClass);
};

const checkInputValidity = (formElement, inputElement, settings) => {
  if (!inputElement.validity.valid) {
    showInputError(formElement, inputElement, inputElement.validationMessage, settings);
  } else {
    hideInputError(formElement, inputElement, settings);
  }
};

const hasInvalidInput = (inputList) => {
  return inputList.some((inputElement) => {
    return !inputElement.validity.valid;
  });
};

const checkDateSequence = (formElement, settings) => {
  const startDate = formElement.querySelector(settings.startDate);
  const endDate = formElement.querySelector(settings.endDate);
  // it works with 1-day vacation
  if (endDate.value && startDate.value > endDate.value) {
    endDate.setCustomValidity("Дата окончания отпуска не может быть меньше даты начала отпуска");
    showInputError(formElement, endDate, settings);
  } else {
    endDate.setCustomValidity("");
    hideInputError(formElement, endDate, settings);
  }
};

const toggleButtonState = (inputList, buttonElement, settings) => {
  if (hasInvalidInput(inputList)) {
    buttonElement.classList.add(settings.inactiveButtonClass);
    buttonElement.disabled = true;
  } else {
    buttonElement.classList.remove(settings.inactiveButtonClass);
    buttonElement.disabled = false;
  }
};

const enableValidation = (formElement, settings) => {
  const inputList = Array.from(formElement.querySelectorAll(settings.inputSelector));
  const buttonElement = formElement.querySelector(settings.submitButtonSelector);
  toggleButtonState(inputList, buttonElement, settings);
  inputList.forEach((inputElement) => {
    inputElement.addEventListener("input", function () {
      checkDateSequence(formElement, settings);
      checkInputValidity(formElement, inputElement, settings);
      toggleButtonState(inputList, buttonElement, settings);
    });
  });
};
